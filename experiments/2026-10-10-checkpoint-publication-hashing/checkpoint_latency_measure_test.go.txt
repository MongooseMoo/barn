package format_test

import (
	"os"
	"path/filepath"
	"strings"
	"testing"
	"time"

	dbformat "github.com/MongooseMoo/barn/db/format"
	"github.com/MongooseMoo/barn/types"
)

// TestMeasureCheckpointLatency times replacing checkpoints of a ~106 MB database.
func TestMeasureCheckpointLatency(t *testing.T) {
	fixture := os.Getenv("CHECKPOINT_FIXTURE")
	if fixture == "" {
		t.Skip("CHECKPOINT_FIXTURE unset")
	}
	database, err := dbformat.LoadDatabase(fixture)
	if err != nil {
		t.Fatal(err)
	}
	objectStore, err := database.NewStoreFromDatabase()
	if err != nil {
		t.Fatal(err)
	}
	chunk := strings.Repeat("x", 100_000)
	padding := make([]types.Value, 1040)
	for i := range padding {
		padding[i] = types.NewStr(chunk)
	}
	objectStore.SetPendingFinalizations(padding)
	path := filepath.Join(t.TempDir(), "measure.db")
	if err := dbformat.WriteCheckpoint(path, objectStore, nil, nil, nil); err != nil {
		t.Fatal(err)
	}
	info, err := os.Stat(path + ".new")
	if err != nil {
		t.Fatal(err)
	}
	t.Logf("checkpoint size %d bytes", info.Size())
	for run := 1; run <= 7; run++ {
		start := time.Now()
		if err := dbformat.WriteCheckpoint(path, objectStore, nil, nil, nil); err != nil {
			t.Fatal(err)
		}
		t.Logf("run %d replacing checkpoint %d ms", run, time.Since(start).Milliseconds())
	}
}
